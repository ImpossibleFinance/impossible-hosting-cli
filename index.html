Install on macOS or Linux

dir="$(mktemp -d)"; trap 'rm -rf "$dir"' EXIT
curl --fail --show-error --location --proto '=https' --proto-redir '=https' --tlsv1.2 \
  https://host.impossibuild.ai/install --output "$dir/install"
curl --fail --show-error --location --proto '=https' --proto-redir '=https' --tlsv1.2 \
  https://host.impossibuild.ai/install.sshsig --output "$dir/install.sshsig"
printf '%s\n' 'ifhost ssh-ed25519 AAAAC3NzaC1lZDI1NTE5AAAAIEv+FR+Wibo0JJPEmmJfqQz2wsoBkrCLatDZ8XwZq2zJ' > "$dir/allowed_signers"
ssh-keygen -Y verify -f "$dir/allowed_signers" -I ifhost -n ifhost-release \
  -s "$dir/install.sshsig" < "$dir/install"
less "$dir/install"
sh "$dir/install"

The trust key above is maintained on this independent release site; it is not supplied by the API host being authenticated.

Install on Windows

$Dir = Join-Path ([IO.Path]::GetTempPath()) "ifhost-$([guid]::NewGuid())"
New-Item -ItemType Directory $Dir | Out-Null
$Installer = Join-Path $Dir 'install.ps1'
$Signature = Join-Path $Dir 'install.ps1.sshsig'
$Allowed = Join-Path $Dir 'allowed_signers'
Invoke-WebRequest https://host.impossibuild.ai/install.ps1 -OutFile $Installer
Invoke-WebRequest https://host.impossibuild.ai/install.ps1.sshsig -OutFile $Signature
[IO.File]::WriteAllText($Allowed, "ifhost ssh-ed25519 AAAAC3NzaC1lZDI1NTE5AAAAIEv+FR+Wibo0JJPEmmJfqQz2wsoBkrCLatDZ8XwZq2zJ`n")
$Verify = 'ssh-keygen -Y verify -f "{0}" -I ifhost -n ifhost-release -s "{1}" < "{2}"' -f $Allowed,$Signature,$Installer
cmd /d /c $Verify
if ($LASTEXITCODE -ne 0) { throw 'Installer signature verification failed' }
Get-Content $Installer
& $Installer
Remove-Item -Recurse -Force $Dir
